## 5. DB 초기화 방법

```sql
-- DB 전체 초기화 후 재생성
DROP DATABASE soma_db;
CREATE DATABASE soma_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE soma_db;
SOURCE /path/to/soma-backend/schema.sql;
```

```sql
-- role ENUM 변경 (기존 DB에 instructor 추가 시)
ALTER TABLE users
  MODIFY COLUMN role ENUM('user','instructor','admin') NOT NULL DEFAULT 'user';

-- lecture_count, total_hours 컬럼 제거 (실시간 계산으로 전환 시)
ALTER TABLE courses
  DROP COLUMN lecture_count,
  DROP COLUMN total_hours;
```
